% Photo booth strip template for NIIMBOT B1 Pro (50x170mm label)
% Film-strip style with rounded frames, perforations, and decorative elements.
%
% This is a PORTRAIT template — design it exactly as the physical strip
% will look when held upright. No rotation is applied during compilation.
%
% Usage:
%   1. Replace photo1.jpg through photo4.jpg with your own images
%   2. Edit the event name, date, and hashtag at the bottom
%   3. Print:  niim-tex --model b1 print photobooth.tex
%
% The \coverphoto macro automatically scales and center-crops each image
% to fill its frame — works with any aspect ratio (portrait, landscape, square).

\documentclass[10pt]{article}
\usepackage[paperwidth=50mm,paperheight=170mm,margin=0mm]{geometry}
\usepackage{tikz}
\usepackage{graphicx}
\usetikzlibrary{calc}
\pagestyle{empty}
\topskip=0pt
\parindent=0pt

% ── Cover-crop macro ─────────────────────────────────────────────────
% \coverphoto{x}{y}{width}{height}{image-path}
%
% Scales the image to FILL the frame (like CSS object-fit:cover), then
% clips the overflow. Automatically chooses width or height scaling
% based on the image's aspect ratio vs the frame's aspect ratio.
\newlength{\coverW}
\newlength{\coverH}
\newcommand{\coverphoto}[5]{%
  \begin{scope}
    \clip[rounded corners=2mm] (#1,#2) rectangle ++(#3,#4);%
    \settowidth{\coverW}{\includegraphics[height=#4mm]{#5}}%
    \ifdim\coverW<#3mm
      \node[inner sep=0pt] at ({#1+#3/2},{#2+#4/2})
        {\includegraphics[width=#3mm]{#5}};%
    \else
      \node[inner sep=0pt] at ({#1+#3/2},{#2+#4/2})
        {\includegraphics[height=#4mm]{#5}};%
    \fi
  \end{scope}
}

% ── Film perforation macro ───────────────────────────────────────────
% Draws a column of rounded rectangles along one edge (like 35mm film)
\newcommand{\perforations}[1]{% #1 = x position of perforation center
  \foreach \py in {4,10,16,...,166} {
    \fill[black] ({#1-1.2},\py) rectangle ++(2.4,3.5);
    \fill[white, rounded corners=0.4mm] ({#1-0.8},{\py+0.3}) rectangle ++(1.6,2.9);
  }
}
