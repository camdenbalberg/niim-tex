% Polaroid-style template for NIIMBOT B1 Pro (50x70mm label)
% Classic instant film look: white border, thick bottom, handwriting-style caption.
%
% This is a PORTRAIT template — design it exactly as the physical label
% will look. No rotation is applied during compilation.
%
% Usage:
%   1. Replace photo.jpg with your image
%   2. Edit the caption text at the bottom
%   3. Print:  niim-tex --model b1 print polaroid.tex
%
% The \coverphoto macro auto-scales and center-crops the image to fill the frame.

\documentclass[10pt]{article}
\usepackage[paperwidth=50mm,paperheight=70mm,margin=0mm]{geometry}
\usepackage{tikz}
\usepackage{graphicx}
\pagestyle{empty}
\topskip=0pt
\parindent=0pt

% ── Cover-crop macro ─────────────────────────────────────────────────
\newlength{\coverW}
\newcommand{\coverphoto}[5]{%
  \begin{scope}
    \clip (#1,#2) rectangle ++(#3,#4);%
    \settowidth{\coverW}{\includegraphics[height=#4mm]{#5}}%
    \ifdim\coverW<#3mm
      \node[inner sep=0pt] at ({#1+#3/2},{#2+#4/2})
        {\includegraphics[width=#3mm]{#5}};%
    \else
      \node[inner sep=0pt] at ({#1+#3/2},{#2+#4/2})
        {\includegraphics[height=#4mm]{#5}};%
    \fi
  \end{scope}
}

\begin{document}
\noindent
\begin{tikzpicture}[x=1mm,y=1mm]
  \useasboundingbox (0,0) rectangle (50,70);

  % ── White background (the polaroid frame) ──────────────────────────
  \fill[white] (0,0) rectangle (50,70);

  % ── Subtle outer shadow / border ──────────────────────────────────
  \draw[black!20, line width=0.3pt] (0.3,0.3) rectangle (49.7,69.7);

  % ── Photo frame ───────────────────────────────────────────────────
  % Polaroid proportions: small side/top margins, thick bottom for caption
  % Side margins: 2mm, Top margin: 2mm, Bottom margin: 12mm
  % Photo area: 46mm wide x 56mm tall
  \coverphoto{2}{12}{46}{56}{photo.jpg}

  % Thin line under photo (subtle, like real Polaroid film edge)
  \draw[black!8, line width=0.2pt] (2,12) -- (48,12);

  % ── Caption area (bottom 12mm) ────────────────────────────────────
  \node[anchor=center, font=\normalsize, text width=44mm, align=center]
    at (25, 6.5) {your caption here};

\end{tikzpicture}
\end{document}
